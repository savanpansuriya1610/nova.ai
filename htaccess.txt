RewriteEngine On
RewriteBase /staging/

# Redirect /staging/page.html -> /staging/page (extensionless)
RewriteCond %{THE_REQUEST} \s/staging/+([^\s?]*?)\.html[\s?] [NC]
RewriteRule ^ %1 [R=301,L]

# Serve extensionless URLs as .html files
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^(.+?)/?$ $1.html [L]
